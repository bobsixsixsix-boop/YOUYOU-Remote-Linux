cmake_minimum_required(VERSION 3.21)
project(youyou_core VERSION 0.1.15 LANGUAGES C CXX)
set(UU_UPSTREAM "" CACHE PATH "Prepared source directory from native/prepare.py")
if(NOT EXISTS "${UU_UPSTREAM}/external/libdatachannel/deps/libsrtp/.fetched")
    message(FATAL_ERROR "First run python3 native/prepare.py --output <directory> and set -DUU_UPSTREAM=<directory>")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(flag ENABLE_TESTING ENABLE_PROGRAMS MBEDTLS_FATAL_WARNINGS JSON_BuildTests
        JSON_Install ENABLE_MBEDTLS ENABLE_OPENSSL ENABLE_WARNINGS_AS_ERRORS
        LIBSRTP_TEST_APPS INSTALL_DEPS_LIBS)
    set(${flag} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(flag NO_WEBSOCKET NO_TESTS NO_EXAMPLES USE_SYSTEM_JSON USE_MBEDTLS)
    set(${flag} ON CACHE BOOL "" FORCE)
endforeach()
include_directories("${UU_UPSTREAM}/include")
add_compile_definitions(MBEDTLS_USER_CONFIG_FILE="uu_mbedtls_config.h")
add_subdirectory("${UU_UPSTREAM}/external/mbedtls" dep-mbed EXCLUDE_FROM_ALL)
add_library(MbedTLS::MbedTLS INTERFACE IMPORTED GLOBAL)
set_target_properties(MbedTLS::MbedTLS PROPERTIES INTERFACE_LINK_LIBRARIES mbedtls
    INTERFACE_INCLUDE_DIRECTORIES "${UU_UPSTREAM}/external/mbedtls/include")
add_library(MbedTLS::MbedCrypto ALIAS mbedcrypto)
add_library(MbedTLS::MbedX509 ALIAS mbedx509)
add_subdirectory("${UU_UPSTREAM}/external/json" dep-json EXCLUDE_FROM_ALL)
set(nlohmann_json_DIR "${CMAKE_BINARY_DIR}/dep-json" CACHE PATH "" FORCE)
add_subdirectory("${UU_UPSTREAM}/external/libdatachannel" dep-rtc EXCLUDE_FROM_ALL)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PLATFORM REQUIRED IMPORTED_TARGET sdl2 libavcodec libavutil libswscale opus libcurl)
pkg_check_modules(GPU REQUIRED IMPORTED_TARGET libva egl glesv2)
find_package(Threads REQUIRED)

# Keep the pinned checkout pristine. Apply only the Linux platform changes to
# the copies we compile. The unified diff is also the source for license review.
find_program(PATCH patch REQUIRED)
set(port "${CMAKE_CURRENT_BINARY_DIR}/linux-source")
file(GLOB upstream_headers CONFIGURE_DEPENDS "${UU_UPSTREAM}/include/*.hpp" "${UU_UPSTREAM}/include/*.h")
foreach(header IN LISTS upstream_headers)
    get_filename_component(name "${header}" NAME)
    configure_file("${header}" "${port}/include/${name}" COPYONLY)
endforeach()
configure_file(capture_rate.hpp "${port}/include/capture_rate.hpp" COPYONLY)
configure_file(decode_catchup.hpp "${port}/include/decode_catchup.hpp" COPYONLY)
configure_file(stream_size.hpp "${port}/include/stream_size.hpp" COPYONLY)
configure_file(virtual_display.hpp "${port}/include/virtual_display.hpp" COPYONLY)
foreach(source virtual_display_wire.inc virtual_display_client.inc)
    configure_file("${source}" "${port}/${source}" COPYONLY)
endforeach()
include_directories(BEFORE "${port}/include")
foreach(file client media transport turn_stream config_store protocol quality_rpc strict_video_receiver stream_stats)
    configure_file("${UU_UPSTREAM}/source/${file}.cpp" "${port}/${file}.cpp" COPYONLY)
endforeach()
foreach(port_patch linux.patch stream_options.patch av1-receiver.patch custom_resolution.patch privacy_screen.patch virtual_display.patch decode_catchup.patch second_screen.patch high_fps.patch zero_copy.patch)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${port_patch}")
    execute_process(COMMAND "${PATCH}" -s -p1 --batch -i "${CMAKE_CURRENT_SOURCE_DIR}/${port_patch}"
        WORKING_DIRECTORY "${port}" COMMAND_ERROR_IS_FATAL ANY)
endforeach()
add_library(youyou_core SHARED bridge.cpp gpu_present.cpp "${port}/media.cpp" "${port}/client.cpp" "${port}/transport.cpp" "${port}/turn_stream.cpp")
foreach(file config_store protocol quality_rpc strict_video_receiver stream_stats)
    target_sources(youyou_core PRIVATE "${port}/${file}.cpp")
endforeach()
foreach(file auth background cpu_diagnostics diagnostics rtc_diagnostics video_assembler)
    target_sources(youyou_core PRIVATE "${UU_UPSTREAM}/source/${file}.cpp")
endforeach()
file(READ "${UU_UPSTREAM}/tests/fixtures/red-16x16.h264" fixture HEX)
string(REGEX REPLACE "(..)" "0x\\1," fixture "${fixture}")
file(WRITE "${port}/test_frame.inc" "${fixture}")
target_include_directories(youyou_core PRIVATE "${port}")
target_compile_definitions(youyou_core PRIVATE UU_VERSION="YOUYOU-Linux-${PROJECT_VERSION}")
target_link_libraries(youyou_core PRIVATE nlohmann_json::nlohmann_json mbedcrypto
    datachannel-static Threads::Threads PkgConfig::PLATFORM PkgConfig::GPU)
target_link_options(youyou_core PRIVATE "-Wl,-z,defs" "-Wl,--exclude-libs,ALL" "-Wl,--wrap=SDL_QueueAudio")
install(TARGETS youyou_core LIBRARY DESTINATION lib)

enable_testing()
add_executable(youyou_decode_catchup_test EXCLUDE_FROM_ALL tests/decode_catchup.cpp)
target_include_directories(youyou_decode_catchup_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME decode_catchup COMMAND youyou_decode_catchup_test)
add_executable(youyou_decode_catchup_media_test EXCLUDE_FROM_ALL tests/decode_catchup_media.cpp)
target_link_libraries(youyou_decode_catchup_media_test PRIVATE youyou_core nlohmann_json::nlohmann_json PkgConfig::PLATFORM ${CMAKE_DL_LIBS})
target_link_options(youyou_decode_catchup_media_test PRIVATE "-Wl,--export-dynamic")
add_executable(youyou_second_screen_test EXCLUDE_FROM_ALL tests/second_screen.cpp)
target_link_libraries(youyou_second_screen_test PRIVATE youyou_core nlohmann_json::nlohmann_json)
add_test(NAME second_screen COMMAND youyou_second_screen_test)
add_executable(youyou_frame_wake_test EXCLUDE_FROM_ALL tests/frame_wake.cpp)
target_link_libraries(youyou_frame_wake_test PRIVATE youyou_core nlohmann_json::nlohmann_json PkgConfig::PLATFORM)
add_test(NAME frame_wake COMMAND youyou_frame_wake_test)
add_executable(youyou_gpu_present_test EXCLUDE_FROM_ALL tests/gpu_present_test.cpp)
target_include_directories(youyou_gpu_present_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(youyou_gpu_present_test PRIVATE youyou_core nlohmann_json::nlohmann_json PkgConfig::PLATFORM)
add_executable(youyou_privacy_test EXCLUDE_FROM_ALL tests/privacy_screen.cpp)
target_link_libraries(youyou_privacy_test PRIVATE youyou_core nlohmann_json::nlohmann_json)
add_test(NAME privacy_screen COMMAND youyou_privacy_test)
add_executable(youyou_capture_options_test EXCLUDE_FROM_ALL tests/capture_options.cpp)
target_link_libraries(youyou_capture_options_test PRIVATE youyou_core nlohmann_json::nlohmann_json)
add_test(NAME capture_options COMMAND youyou_capture_options_test)
add_executable(youyou_virtual_display_test EXCLUDE_FROM_ALL tests/virtual_display.cpp)
target_link_libraries(youyou_virtual_display_test PRIVATE youyou_core nlohmann_json::nlohmann_json)
add_test(NAME virtual_display COMMAND youyou_virtual_display_test)
add_executable(youyou_decode_probe EXCLUDE_FROM_ALL tests/decode_probe.cpp)
target_include_directories(youyou_decode_probe PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(youyou_decode_probe PRIVATE youyou_core nlohmann_json::nlohmann_json PkgConfig::PLATFORM datachannel-static)
add_executable(youyou_rtc_loopback EXCLUDE_FROM_ALL "${UU_UPSTREAM}/tests/rtc_test.cpp"
    "${UU_UPSTREAM}/source/background.cpp" "${UU_UPSTREAM}/source/diagnostics.cpp"
    "${port}/protocol.cpp" "${UU_UPSTREAM}/source/rtc_diagnostics.cpp")
target_compile_definitions(youyou_rtc_loopback PRIVATE UU_VERSION="YOUYOU-Linux-test")
target_link_libraries(youyou_rtc_loopback PRIVATE nlohmann_json::nlohmann_json mbedcrypto datachannel-static Threads::Threads)
add_test(NAME rtc_loopback COMMAND youyou_rtc_loopback)
set_tests_properties(rtc_loopback PROPERTIES TIMEOUT 45)
include(overlay.cmake)
